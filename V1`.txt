function tileAt(x,y){
  const tx=Math.floor(x/T),ty=Math.floor(y/T);
  if(tx<0||ty<0||tx>=MW||ty>=MH)return 0;
  return map[ty][tx];
}
function freeAt(x,y,r){return tileAt(x-r,y-r)&&tileAt(x+r,y-r)&&tileAt(x-r,y+r)&&tileAt(x+r,y+r)}
function slide(e,dx,dy,r){
  const steps=Math.max(1,Math.ceil(Math.hypot(dx,dy)/5));
  const sx=dx/steps,sy=dy/steps;
  for(let i=0;i<steps;i++){
    if(freeAt(e.x+sx,e.y,r))e.x+=sx;
    if(freeAt(e.x,e.y+sy,r))e.y+=sy;
  }
}
function roomPos(r){
  for(let i=0;i<30;i++){
    const tx=ri(r.x+1,r.x+r.w-2),ty=ri(r.y+1,r.y+r.h-2);
    if(map[ty][tx]===1)return{x:tx*T+T/2,y:ty*T+T/2};
  }
  return{x:(r.x+r.w/2)*T,y:(r.y+r.h/2)*T};
}

function genFloor(){
  map=Array.from({length:MH},()=>new Uint8Array(MW));
  seen=Array.from({length:MH},()=>new Uint8Array(MW));
  rooms=[];chests=[];enemies=[];bullets=[];ebullets=[];ground=[];

    if(rooms.some(r=>x<r.x+r.w+2&&x+w+2>r.x&&y<r.y+r.h+2&&y+h+2>r.y))continue;
    rooms.push({x,y,w,h});
    for(let yy=y;yy<y+h;yy++)for(let xx=x;xx<x+w;xx++)map[yy][xx]=1;
  }
  const cen=r=>({x:Math.floor(r.x+r.w/2),y:Math.floor(r.y+r.h/2)});
  for(let i=1;i<rooms.length;i++){
    const a=cen(rooms[i-1]),b=cen(rooms[i]);
    let x=a.x;while(x!==b.x){map[a.y][x]=1;x+=x<b.x?1:-1}
    let y=a.y;while(y!==b.y){map[y][b.x]=1;y+=y<b.y?1:-1}
  }
  const last=rooms.length-1;
  for(let i=1;i<last;i++){
    if(Math.random()<0.55){
      const r=rooms[i],cx=r.x+r.w/2,cy=r.y+r.h/2,rad=Math.min(r.w,r.h)/2.6;
      for(let yy=r.y;yy<r.y+r.h;yy++)for(let xx=r.x;xx<r.x+r.w;xx++){
        if(map[yy][xx]===1&&Math.hypot(xx+.5-cx,yy+.5-cy)<rad+rnd(-.6,.6))map[yy][xx]=2;
      }
    }
  }
  const L=rooms[last];
  stairsT={x:L.x+L.w-2,y:L.y+L.h-2};
  map[stairsT.y][stairsT.x]=3;
  for(let i=1;i<last;i++){
    if(Math.random()<0.5){
      const p=roomPos(rooms[i]);
      chests.push({x:p.x,y:p.y,open:false});

  const s0=rooms[0];
  P.x=(s0.x+s0.w/2)*T;P.y=(s0.y+s0.h/2)*T;P.vx=0;P.vy=0;
}

}

// ---------- Combat ----------
function dealDamage(e,base){
  if(e.dead)return;
  let dmg=base,crit=false;
  if(Math.random()*100<S.critC){dmg*=S.critM/100;crit=true}
  e.hp-=dmg;e.flash=0.08;
  addText(e.x,e.y-e.r-6,Math.round(dmg),crit?'#ffe14f':'#ffffff',crit?1.5:1);
  if(S.leech)P.hp=Math.min(P.maxHp,P.hp+dmg*S.leech/100);
  if(hasTrait('siphon'))P.sh=Math.min(P.maxSh,P.sh+dmg*0.15);
  burst(e.x,e.y,e.c,3);
  if(e.hp<=0)killEnemy(e);
}
function killEnemy(e){
  if(e.dead)return;
  e.dead=true;
  const Tt=ET[e.type];
  P.xp+=Tt.xp*(1+floor*0.1);
  while(P.xp>=P.xpNext){
    P.xp-=P.xpNext;P.lvl++;P.xpNext=Math.round(P.xpNext*1.35);
    calc();P.hp=P.maxHp;P.sh=P.maxSh;
    pendingLv++;
  }
  if(pendingLv>0&&!lvOpen)openLevelUp();
  stardust+=ri(1,3);
  burst(e.x,e.y,e.c,18);
  if(hasTrait('inferno')){
    burst(e.x,e.y,'#ff7a2b',24);
    for(const o of enemies){
      if(o!==e&&!o.dead&&dist(o,e)<100)dealDamage(o,S.damage*0.6);
    }
  }
  if(e.boss){
    burst(e.x,e.y,'#ff2b6b',80);
    shake=1;
    for(let k=0;k<3;k++)dropItem(e.x,e.y,3,floor+3);
    announce('THE WARDEN FALLS','#ffe14f',2.5);
    log('The stairs are unsealed.','#ffe14f');
  }else if(Math.random()<Tt.drop){
    dropItem(e.x,e.y,e.type==='demon'?2:0,floor+ri(0,3));
  }
}
function hurt(d,noIF){
  if(dead)return;
  if(!noIF&&P.inv>0)return;
  P.sinceHit=0;
  let rem=d;
  if(P.sh>0){const a=Math.min(P.sh,rem);P.sh-=a;rem-=a}
  P.hp-=rem;shake=Math.max(shake,0.25);
  addText(P.x,P.y-24,'-'+Math.round(d),'#ff4f6e',1);
  if(P.hp<=0){
    P.hp=0;dead=true;
    announce('THE KNIGHT FALLS','#ff4f6e',99);
    log('Press R to begin again','#ff4f6e');
  }
}
function swing(){
  P.atk=1/S.atkRate;P.swings++;
  const oath=hasTrait('oath')&&P.swings%5===0;
  swings.push({x:P.x,y:P.y,a:P.aim,life:.18,r:S.range,arc:S.arcDeg*Math.PI/180,oath});
  const half=S.arcDeg*Math.PI/360;
  for(const e of enemies){
    if(e.dead)continue;
    const d=dist(e,P);
    if(d>S.range+e.r)continue;
    let da=Math.atan2(e.y-P.y,e.x-P.x)-P.aim;
    da=Math.atan2(Math.sin(da),Math.cos(da));
    if(Math.abs(da)<=half+Math.asin(Math.min(1,e.r/Math.max(d,1))))dealDamage(e,S.damage);
  }
  if(oath){
    for(let k=0;k<12;k++){
      const a=k/12*TAU;
      bullets.push({x:P.x,y:P.y,vx:Math.cos(a)*420,vy:Math.sin(a)*420,life:.8,dmg:S.damage*0.6,left:1,hit:[],c:'#ff7a2b'});
    }
    burst(P.x,P.y,'#ff7a2b',20);
  }
}

// ---------- Update ----------
function update(dt){
  time+=dt;
  updateFX(dt);
  if(Math.random()<dt*8){
    parts.push({x:P.x+rnd(-300,300),y:P.y+rnd(-300,300),vx:rnd(-10,10),vy:rnd(-40,-10),life:rnd(1,2.5),max:2.5,c:pick(['#ff7a2b','#ffb347','#ff4f2b']),s:rnd(1,2.5)});
  }
  if(invOpen||dead||lvOpen)return;

  const mx=(keys.d||keys.arrowright?1:0)-(keys.a||keys.arrowleft?1:0);
  const my=(keys.s||keys.arrowdown?1:0)-(keys.w||keys.arrowup?1:0);
  const ml=Math.hypot(mx,my)||1;
  const ux=mx/ml,uy=my/ml;
  const tvx=(mx||my)?ux*S.spd:0,tvy=(mx||my)?uy*S.spd:0;
  P.vx+=(tvx-P.vx)*Math.min(1,dt*10);
  P.vy+=(tvy-P.vy)*Math.min(1,dt*10);
  slide(P,P.vx*dt,P.vy*dt,10);
  P.aim=Math.atan2(mouse.y-H/2,mouse.x-W/2);

  P.roll-=dt;P.inv-=dt;P.atk-=dt;P.sinceHit+=dt;
  if(keys[' ']&&P.roll<=0){
    let dx=ux,dy=uy;
    if(!mx&&!my){dx=Math.cos(P.aim);dy=Math.sin(P.aim)}
    slide(P,dx*130,dy*130,10);
    P.roll=S.rollCD;P.inv=0.3;
    burst(P.x,P.y,'#ffb347',20);
    keys[' ']=false;
  }
  if(mouse.down&&P.atk<=0)swing();

  if(P.sinceHit>3)P.sh=Math.min(P.maxSh,P.sh+P.maxSh*0.2*dt);
  if(P.sinceHit>4)P.hp=Math.min(P.maxHp,P.hp+S.regen*dt);

  if(tileAt(P.x,P.y)===2)hurt(16*dt,true);

  if(tileAt(P.x,P.y)===3){
    if(boss&&!boss.dead){
      if(!P.warned){log('The Warden seals the stairs. Slay him!','#ff4f6e');P.warned=true}
    }else{
      descend();return;
    }
  }else P.warned=false;

  // reveal fog
  const ptx=Math.floor(P.x/T),pty=Math.floor(P.y/T);
  for(let dy=-8;dy<=8;dy++)for(let dx=-8;dx<=8;dx++){
    const x=ptx+dx,y=pty+dy;
    if(dx*dx+dy*dy<=64&&x>=0&&y>=0&&x<MW&&y<MH)seen[y][x]=1;
  }

  // enemies
  for(const e of enemies){
    if(e.dead)continue;
    const Tt=ET[e.type];
    const dx=P.x-e.x,dy=P.y-e.y,d=Math.hypot(dx,dy)||1;
    const nx=dx/d,ny=dy/d;
    e.touch-=dt;e.flash-=dt;
    let vx=0,vy=0;
    if(d<Tt.aggro||e.boss){
      if(e.type==='archer'){
        if(d>170){vx=nx*e.spd;vy=ny*e.spd}
        else if(d<120){vx=-nx*e.spd;vy=-ny*e.spd}
        e.fire-=dt;
        if(e.fire<=0&&d<620){
          ebullets.push({x:e.x,y:e.y,vx:nx*190,vy:ny*190,life:3.5,dmg:9+floor*2,r:4,c:'#ffe6b0'});
          e.fire=Tt.shoot;
        }
      }else if(e.boss){
        vx=nx*e.spd;vy=ny*e.spd;
        e.shot-=dt;
        if(e.shot<=0){
          e.shot=2.4;
          for(let k=0;k<14;k++){
            const a=k/14*TAU+e.phase;
            ebullets.push({x:e.x,y:e.y,vx:Math.cos(a)*170,vy:Math.sin(a)*170,life:4,dmg:12+floor*2,r:6,c:'#ff2b6b'});
          }
          e.phase+=0.2;
        }
      }else{
        vx=nx*e.spd;vy=ny*e.spd;
      }
    }else{
      e.wt-=dt;
      if(e.wt<=0){e.wt=rnd(1,3);e.wa=rnd(0,TAU)}
      vx=Math.cos(e.wa)*e.spd*.3;vy=Math.sin(e.wa)*e.spd*.3;
    }
    slide(e,vx*dt,vy*dt,e.r*0.8);
    if(tileAt(e.x,e.y)===2){
      e.hp-=14*dt;
      if(e.hp<=0){killEnemy(e);continue}
    }
    if(d<e.r+10&&e.touch<=0){
      const mult=(1+(floor-1)*0.15)*(e.boss?1.2:1);
      hurt(Tt.dmg*mult);
      e.touch=0.7;
      burst(e.x,e.y,'#ff4f2b',6);
    }
  }

  for(const b of bullets){
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    for(const e of enemies){
      if(b.left<=0)break;
      if(e.dead||b.hit.includes(e))continue;
      if(Math.hypot(e.x-b.x,e.y-b.y)<e.r+4){b.hit.push(e);b.left--;dealDamage(e,b.dmg)}
    }
    if(b.left<=0)b.life=0;
  }
  bullets=bullets.filter(b=>b.life>0);

  for(const b of ebullets){
    b.x+=b.vx*dt;b.y+=b.vy*dt;b.life-=dt;
    if(tileAt(b.x,b.y)===0){b.life=0;continue}
    if(Math.hypot(b.x-P.x,b.y-P.y)<b.r+10){hurt(b.dmg);b.life=0}
  }
  ebullets=ebullets.filter(b=>b.life>0);

  for(const s of swings)s.life-=dt;
  swings=swings.filter(s=>s.life>0);

  for(const g of ground){
    g.age+=dt;g.bob+=dt*3;
    if(g.r<filter)continue;
    const d=dist(g,P);
    if(d<S.mag){g.x+=(P.x-g.x)*Math.min(1,dt*10);g.y+=(P.y-g.y)*Math.min(1,dt*10)}
    if(d<18&&inv.length<60){
      inv.push(g);g.dead=true;
      log(`Picked up ${g.name}`,colorOf(g));
      if(g.r>=4)announce(RAR[g.r].n.toUpperCase(),colorOf(g),1.4);
      if(invOpen)renderInv();
    }
  }
  ground=ground.filter(g=>!g.dead&&g.age<120);
  enemies=enemies.filter(e=>!e.dead);
  calc();
}

function shape(type,x,y,r){
  ctx.beginPath();
  if(type==='circle')ctx.arc(x,y,r,0,TAU);
  else if(type==='diamond'){ctx.moveTo(x,y-r);ctx.lineTo(x+r,y);ctx.lineTo(x,y+r);ctx.lineTo(x-r,y);ctx.closePath()}
  else{for(let i=0;i<6;i++){const a=i/6*TAU+0.5,px=x+Math.cos(a)*r,py=y+Math.sin(a)*r;i?ctx.lineTo(px,py):ctx.moveTo(px,py)}ctx.closePath()}
}

function openChest(){
  let best=null,bd=44;
  for(const c of chests){const d=dist(c,P);if(!c.open&&d<bd){bd=d;best=c}}
  if(!best)return;
  best.open=true;
  for(let k=0;k<ri(2,3);k++)dropItem(best.x,best.y,1,floor+2);
  burst(best.x,best.y,'#ffe14f',30);
  log('Chest opened.','#ffe14f');
}
